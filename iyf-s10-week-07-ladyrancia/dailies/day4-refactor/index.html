What's Wrong?

  • Unclear variable names (u, e, n, a, d)
  • Magic numbers (0.1, 10)

  • Mixed concerns (validation, DB, UI all in one)
  • No comments or documentation
  • Uses var instead of const/let
  • Hard-coded strings
  • Poor formatting

AFTER Clean, Refactored Code 

// Constants
const MIN_AGE = 18;
const DEFAULT_DISCOUNT_RATE = 0.1;
const MAX_USERS = 10;

/**
 * Validate user data
 * @param {Object} userData - User information
 * @throws {Error} If validation fails
 */
function validateUserData(userData) {
    if (!userData.email.includes("@")) {
        throw new Error("Invalid email address");
    }
    if (userData.age < MIN_AGE) {
        throw new Error("User must be at least " + MIN_AGE + " years old");
    }
}

/**
 * Normalize user data
 * @param {Object} userData - Raw user data
 * @returns {Object} Normalized user data
 */
function normalizeUserData(userData) {
    return {
        name: userData.name.trim(),
        email: userData.email.toLowerCase(),
        age: userData.age
    };
}

/**
 * Save user to database
 * @param {Object} user - Normalized user object
 * @returns {Promise} Save operation
 */
async function saveUserToDatabase(user) {
    return await db.save({
        name: user.name,
        email: user.email,
        age: user.age
    });
}

/**
 * Send welcome email
 * @param {string} email - User email
 */
async function sendWelcomeEmail(email) {
    await sendEmail(email, "Welcome to our platform!");
}

/**
 * Update UI after successful user creation
 */
function updateSuccessUI() {
    const statusEl = document.getElementById("status");
    if (statusEl) {
        statusEl.textContent = "User created successfully!";
        statusEl.className = "success";
    }
}

/**
 * Main function to process user registration
 * @param {Object} userData - Raw user data from form
 * @returns {Object} Result object
 */
async function processUserRegistration(userData) {
    try {
        // Validate input
        validateUserData(userData);

        // Normalize data
        const normalizedUser = normalizeUserData(userData);

        // Persist to database
        await saveUserToDatabase(normalizedUser);

        // Send welcome email (async, fire-and-forget)
        sendWelcomeEmail(normalizedUser.email).catch(console.error);

        // Update UI
        updateSuccessUI();

        return { success: true, user: normalizedUser };
    } catch (error) {
        console.error("User registration failed:", error);
        document.getElementById("status").textContent = "Error: " + error.message;
        document.getElementById("status").className = "error";
        return { success: false, error: error.message };
    }
}

/**
 * Generate sample users
 * @param {number} count - Number of users to generate
 * @returns {Array} Array of user objects
 */
function generateSampleUsers(count = MAX_USERS) {
    const users = [];
    for (let i = 0; i < count; i++) {
        users.push({
            id: i,
            name: `User${i}`,
            email: `user${i}@test.com`,
            age: 20 + i
        });
    }
    return users;
}

/**
 * Calculate discount total
 * @param {number} basePrice - Base price
 * @param {number} quantity - Quantity
 * @param {number} discountRate - Discount rate (default: DEFAULT_DISCOUNT_RATE)
 * @returns {number} Final price after discount
 */
function calculateDiscountedTotal(basePrice, quantity, discountRate = DEFAULT_DISCOUNT_RATE) {
    const subtotal = basePrice * quantity;
    return subtotal * (1 - discountRate);
}

// Usage examples
const sampleUsers = generateSampleUsers();
const discountedTotal = calculateDiscountedTotal(100, 50, 3);
console.log("Discounted total:", discountedTotal);

Improvements Made

  • Meaningful variable and function names
  • Single Responsibility Principle - each function does ONE thing
  • Constants instead of magic numbers
  • Proper error handling with detailed messages
  • JSDoc comments for documentation
  • Uses const/let instead of var
  • Separated concerns: validation, normalization, persistence, UI
  • Async/await for promise handling
  • Returns consistent result objects

📚 Refactoring Tips

1. Extract Meaningful Names

// Before
var d = new Date();
var x = data.filter(u => u.a > 18);

// After
const currentDate = new Date();
const adultUsers = users.filter(user => user.age > 18);

2. Single Responsibility

// Before - Does too many things
function processOrder(order) {
    validate(order);
    calculateTotal(order);
    saveToDB(order);
    sendEmail(order);
    updateUI();
}

// After - Each function has ONE job
function validateOrder(order) { /* validation only */ }
function calculateOrderTotal(order) { /* calculation only */ }
async function persistOrder(order) { /* database only */ }
async function notifyCustomer(order) { /* email only */ }
function renderOrderConfirmation(order) { /* UI only */ }

3. Replace Magic Numbers

// Before
if (password.length < 8) { }
setTimeout(callback, 86400000);

// After
const MIN_PASSWORD_LENGTH = 8;
const ONE_DAY_MS = 24 * 60 * 60 * 1000;
if (password.length < MIN_PASSWORD_LENGTH) { }
setTimeout(callback, ONE_DAY_MS);

4. Error Handling

// Before - Silent failures
function loadUser(id) {
    return db.find(id);  // What if not found?
}

// After - Proper error handling
async function loadUser(id) {
    if (!id) {
        throw new Error("User ID is required");
    }
    try {
        const user = await db.find(id);
        if (!user) {
            throw new Error("User not found");
        }
        return user;
    } catch (error) {
        console.error("Failed to load user:", error);
        throw error;  // Re-throw for caller to handle
    }
}
